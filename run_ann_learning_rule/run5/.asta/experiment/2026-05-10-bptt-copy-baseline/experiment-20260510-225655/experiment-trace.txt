Top-Level Task: BACKGROUND KNOWLEDGE FOR STEP 3: BPTT baseline on copy task

TASK
----
Run a BPTT baseline experiment on a short supervised copy task using a vanilla RNN,
so that downstream steps (implementing RFLO, e-prop, and STC-Credit from design_step2.md)
have a reference performance target.

PURPOSE
-------
`design_step2.md` specified STC-Credit, the candidate biologically-plausible rule for H1.
Before implementing it, we need a well-characterized BPTT baseline on the same architecture
and task. Success criterion (from mission.md): "Performance on short-sequence supervised
tasks is within reasonable range of BPTT" — so we must first quantify BPTT's performance.

TASK SPEC — COPY TASK (length 10)
----------------------------------
Input: At t = 1..5, 5 symbols drawn uniformly from an alphabet of 8 (one-hot encoded,
so input dim = 8 + 2 = 10; the extra 2 channels are "blank" and "delimiter").
From t = 6..10 the input is "blank".
At t = 10 a "go" signal appears (delimiter channel).
Target: output the original 5 symbols at t = 11..15 (output dim = 8).

This is the canonical short copy task used for RFLO / e-prop comparisons and is simple
enough that a small RNN trained with BPTT should learn it cleanly, giving us a non-degenerate
reference. Total sequence length = 15 timesteps.

EXPERIMENT CONFIGURATION
------------------------
- Architecture: vanilla (Elman) RNN
  - Hidden size: 128 (slightly larger than the 64 in design_step2 §7 defaults —
    128 gives cleaner convergence for BPTT baselines and makes the BPTT error floor
    easy to establish).
  - Activation: tanh
  - Input dim: 10 (8 symbols + blank + go)
  - Output dim: 8 (over symbols only)
  - Readout at every timestep; cross-entropy loss only applied on output timesteps t=11..15.
- Optimizer: Adam, lr = 1e-3
- Batch size: 64
- Number of training steps: 3000 (should easily saturate on this task)
- Evaluation: every 100 steps, on 1000 held-out sequences
- Metrics:
  - Per-symbol accuracy on the output timesteps
  - Sequence-level accuracy (all 5 symbols correct)
  - Cross-entropy loss
  - Wall-clock time per 100 training steps

DEVICE HANDLING
---------------
- Detect GPU at runtime via torch.cuda.is_available()
- Move model + batches to GPU if present, otherwise CPU
- Log the device at start
- Code must work on both

  "Train a vanilla Elman RNN with backpropagation-through-time (BPTT) on a standard
   copy task where a 5-symbol sequence from an alphabet of size 8 is presented during
   timesteps 1..5, followed by 5 blank timesteps and a single 'go' cue at timestep 10,
   and the network must reproduce the 5 symbols at timesteps 11..15. Report the per-symbol
   classification accuracy, sequence-level accuracy, and cross-entropy loss learning curve.
   Use Adam with lr=1e-3, batch size 64, hidden size 128, tanh activation. Run for 3000
   training steps with evaluation every 100 steps on 1000 held-out sequences. Use GPU if
   available (torch.cuda.is_available()); fall back to CPU. Log the device. Do not use
   any built-in PyTorch RNN module — implement the recurrent dynamics explicitly as

import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
import time

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f"Device: {device}")
if torch.cuda.is_available():
    print(f"GPU: {torch.cuda.get_device_name(0)}")

INPUT_DIM = 10
HIDDEN_DIM = 128
OUTPUT_DIM = 8
ALPHABET = 8
SEQ_LEN = 15
N_SYMBOLS = 5
BATCH_SIZE = 64
LR = 1e-3
N_STEPS = 3000
EVAL_EVERY = 100
EVAL_SIZE = 1000
BLANK_CH = 8
GO_CH = 9
OUT_START = 10

def generate_batch(batch_size, device):
    symbols = torch.randint(0, ALPHABET, (batch_size, N_SYMBOLS))
    x = torch.zeros(batch_size, SEQ_LEN, INPUT_DIM)
    for i in range(N_SYMBOLS):
        x[torch.arange(batch_size), i, symbols[:, i]] = 1.0
    x[:, 5:9, BLANK_CH] = 1.0
    x[:, 9, GO_CH] = 1.0
    x[:, 9, BLANK_CH] = 1.0
    x[:, 10:15, BLANK_CH] = 1.0
    return x.to(device), symbols.to(device)

class ElmanRNN(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        super().__init__()
        self.hidden_dim = hidden_dim
        self.W = nn.Parameter(torch.randn(hidden_dim, hidden_dim) * 0.1)
        self.U = nn.Parameter(torch.randn(hidden_dim, input_dim) * 0.1)
        self.b_h = nn.Parameter(torch.zeros(hidden_dim))
        self.V = nn.Parameter(torch.randn(output_dim, hidden_dim) * 0.1)
        self.b_y = nn.Parameter(torch.zeros(output_dim))

    def forward(self, x):
        batch_size, T, _ = x.shape
        h = torch.zeros(batch_size, self.hidden_dim, device=x.device)
        output_logits = []
        for t in range(T):
            x_t = x[:, t, :]
            h = torch.tanh(h @ self.W.T + x_t @ self.U.T + self.b_h)
            if t >= OUT_START:
                output_logits.append(h @ self.V.T + self.b_y)
        return torch.stack(output_logits, dim=1)

@torch.no_grad()
def evaluate(model):
    model.eval()
    x, targets = generate_batch(EVAL_SIZE, device)
    logits = model(x)
    loss = nn.functional.cross_entropy(logits.reshape(-1, OUTPUT_DIM), targets.reshape(-1)).item()
    preds = logits.argmax(dim=-1)
    per_sym = (preds == targets).float().mean().item()
    seq_acc = (preds == targets).all(dim=1).float().mean().item()
    model.train()
    return loss, per_sym, seq_acc

print("="*70)
print("BPTT BASELINE: Vanilla Elman RNN on Copy Task")
print("="*70)
print(f"Architecture: input={INPUT_DIM}, hidden={HIDDEN_DIM}, output={OUTPUT_DIM}")
print(f"Training: Adam lr={LR}, batch={BATCH_SIZE}, steps={N_STEPS}")
print("="*70)

model = ElmanRNN(INPUT_DIM, HIDDEN_DIM, OUTPUT_DIM).to(device)
optimizer = optim.Adam(model.parameters(), lr=LR)
loss_fn = nn.CrossEntropyLoss()

results = []
step_times = []

loss0, sym0, seq0 = evaluate(model)
results.append({"step":0,"loss":loss0,"per_symbol_acc":sym0,"seq_acc":seq0})
print(f"Step    0 | Loss: {loss0:.4f} | Per-sym: {sym0:.4f} | Seq: {seq0:.4f}")

seg_start = time.time()
for step in range(1, N_STEPS+1):
    model.train()
    x, targets = generate_batch(BATCH_SIZE, device)
    optimizer.zero_grad()
    logits = model(x)
    loss = loss_fn(logits.reshape(-1, OUTPUT_DIM), targets.reshape(-1))
    loss.backward()
    optimizer.step()
    if step % EVAL_EVERY == 0:
        elapsed = time.time() - seg_start
        step_times.append(elapsed)
        el, sym, seq = evaluate(model)
        results.append({"step":step,"loss":el,"per_symbol_acc":sym,"seq_acc":seq,"wall_time":elapsed})
        print(f"Step {step:4d} | Loss: {el:.4f} | Per-sym: {sym:.4f} | Seq: {seq:.4f} | Time: {elapsed:.2f}s")
        seg_start = time.time()

print("="*70)
final = results[-1]
print(f"FINAL: step={final[\"step\"]}, per_sym={final[\"per_symbol_acc\"]:.4f}, seq={final[\"seq_acc\"]:.4f}, loss={final[\"loss\"]:.4f}")
print(f"Avg time/100 steps: {np.mean(step_times):.2f}s")
print(f"Device: {device}")

landmark = None
for r in results:
    if r["per_symbol_acc"] > 0.90:
        landmark = r["step"]
        break
if landmark is not None:
    print(f"LANDMARK: per-symbol acc first >90% at step {landmark}")
else:
    print("LANDMARK: per-symbol acc did NOT exceed 90% within 3000 steps")

print(f"Success (>95% per-sym): {final[\"per_symbol_acc\"] > 0.95}")
print("="*70)
print("LEARNING CURVE:")
print("step | loss | per_sym_acc | seq_acc")
for r in results:
    print(f"{r[\"step\"]:4d} | {r[\"loss\"]:.4f} | {r[\"per_symbol_acc\"]:.4f} | {r[\"seq_acc\"]:.4f}")
'''
